def apply\_spell(domain, role, all\_domains=None):  
    \# local & global can both matter  
    if role \== "heal":  
        domain.health \= min(1.0, domain.health \+ 0.03)  
        domain.load \= max(0.0, domain.load \- 0.03)  
    elif role \== "heal\_strong":  
        domain.health \= min(1.0, domain.health \+ 0.06)  
        domain.load \= max(0.0, domain.load \- 0.06)  
    elif role \== "preserve":  
        \# resist deterioration  
        domain.health \= min(1.0, domain.health \+ 0.02)  
    elif role \== "balance":  
        \# balancing is done at multi-domain level; flag via health  
        domain.health \= min(1.0, domain.health \+ 0.01)  
    elif role \== "feedback":  
        \# adjust load based on past state (here: small stabilization)  
        domain.load \= (domain.load \+ 0.5) / 2  
    elif role \== "fairness":  
        \# small boost to trust  
        domain.trust \= min(1.0, domain.trust \+ 0.03)  
    elif role \== "insight":  
        domain.load \+= random.uniform(-0.02, 0.02)  
    elif role \== "predict":  
        \# predictive awareness reduces surprise load spikes  
        domain.load \= max(0.0, domain.load \- 0.02)  
    elif role \== "monitor":  
        \# improves capacity to respond  
        domain.capacity \= min(2.0, domain.capacity \+ 0.01)  
    elif role \== "depth":  
        \# deepen capacity slowly  
        domain.capacity \= min(2.0, domain.capacity \+ 0.02)  
    elif role \== "resource":  
        \# better resource handling  
        domain.capacity \= min(2.0, domain.capacity \+ 0.03)  
    elif role \== "boost":  
        \# temporary overclock, more capacity but more load  
        domain.capacity \= min(2.5, domain.capacity \+ 0.05)  
        domain.load \= min(1.0, domain.load \+ 0.03)  
    elif role \== "shield":  
        \# resist load increase  
        domain.load \= max(0.0, domain.load \- 0.03)  
    elif role \== "absorb":  
        domain.load \= max(0.0, domain.load \- 0.05)  
    elif role \== "impact":  
        \# big action: reduces others' load but self load rises  
        domain.load \= min(1.0, domain.load \+ 0.04)  
        if all\_domains:  
            for d in all\_domains:  
                if d is not domain:  
                    d.load \= max(0.0, d.load \- 0.02)  
    elif role \== "meta":  
        domain.capacity \= min(3.0, domain.capacity \+ 0.03)  
        domain.trust \= min(1.0, domain.trust \+ 0.02)  
    elif role \== "unify":  
        if all\_domains:  
            for d in all\_domains:  
                d.trust \= min(1.0, d.trust \+ 0.02)  
    elif role \== "connect":  
        domain.trust \= min(1.0, domain.trust \+ 0.04)  
    elif role \== "relate":  
        domain.trust \= min(1.0, domain.trust \+ 0.03)  
    elif role \== "broadcast":  
        if all\_domains:  
            for d in all\_domains:  
                d.trust \= min(1.0, d.trust \+ 0.01)  
    elif role \== "message":  
        domain.trust \= min(1.0, domain.trust \+ 0.02)  
    \# neutral: do nothing

def apply\_chain(domain, chain, all\_domains):  
    for spell in chain:  
        role \= SPELL\_ROLES.get(spell, DEFAULT\_ROLE)  
        apply\_spell(domain, role, all\_domains=all\_domains)

def run\_hive\_sim(domains, bridges, hours=24):  
    print("=== INITIAL STATE \===")  
    for d in domains:  
        print(d)  
    print()

    for hour in range(hours):  
        print(f"\\n=== HOUR {hour} \===")

        \# Crisis progression example: randomly stress some domains  
        for d in domains:  
            d.load \= min(1.0, max(0.0, d.load \+ random.uniform(-0.02, 0.04)))  
            d.health \= max(0.0, d.health \- random.uniform(0.0, 0.01))

        \# Apply domain-local chains  
        for d in domains:  
            for chain in d.chains:  
                apply\_chain(d, chain, domains)

        \# Apply bridges (cross-domain effects)  
        for br in bridges:  
            a \= next(d for d in domains if d.name \== br.domain\_a)  
            b \= next(d for d in domains if d.name \== br.domain\_b)  
            \# symmetric effect for now  
            for spell in br.chain:  
                role \= SPELL\_ROLES.get(spell, DEFAULT\_ROLE)  
                apply\_spell(a, role, all\_domains=domains)  
                apply\_spell(b, role, all\_domains=domains)

        \# Simple global balancer: like Koantra/Equilibria  
        max\_load \= max(d.load for d in domains)  
        min\_load \= min(d.load for d in domains)  
        if max\_load \- min\_load \> 0.4:  
            avg \= sum(d.load for d in domains) / len(domains)  
            for d in domains:  
                d.load \= (d.load \+ avg) / 2

        for d in domains:  
            print(d)

    print("\\n=== FINAL STATE \===")  
    for d in domains:  
        print(d)

class DomainState:  
    def \_\_init\_\_(self, name, cluster, wrap, chains):  
        self.name \= name  
        self.cluster \= cluster  
        self.wrap \= wrap  
        self.chains \= chains  \# list of lists of spell names  
        self.load \= 0.5  
        self.capacity \= 1.0  
        self.trust \= 0.5  
        self.health \= 1.0  \# 1.0 \= optimal, 0 \= collapse

    def \_\_repr\_\_(self):  
        return (f"{self.name}: load={self.load:.2f}, cap={self.capacity:.2f}, "  
                f"trust={self.trust:.2f}, health={self.health:.2f}")

class Bridge:  
    def \_\_init\_\_(self, domain\_a, domain\_b, chain):

        self.domain\_b \= domain\_b  
        self.chain \= chain  \# list of spell names

def apply\_spell(domain, role, all\_domains=None):  
    if role \== "heal":  
        domain.health \= min(1.0, domain.health \+ 0.03)  
        domain.load \= max(0.0, domain.load \- 0.03)  
    elif role \== "heal\_strong":  
        domain.health \= min(1.0, domain.health \+ 0.06)  
        domain.load \= max(0.0, domain.load \- 0.06)  
    elif role \== "preserve":  
        domain.health \= min(1.0, domain.health \+ 0.02)  
    elif role \== "balance":  
        domain.health \= min(1.0, domain.health \+ 0.01)  
    elif role \== "feedback":  
        domain.load \= (domain.load \+ 0.5) / 2  
    elif role \== "fairness":  
        domain.trust \= min(1.0, domain.trust \+ 0.03)  
    elif role \== "insight":  
        domain.load \= max(0.0, min(1.0, domain.load \+ random.uniform(-0.02, 0.02)))  
    elif role \== "predict":  
        domain.load \= max(0.0, domain.load \- 0.02)  
    elif role \== "monitor":  
        domain.capacity \= min(2.0, domain.capacity \+ 0.01)  
    elif role \== "depth":  
        domain.capacity \= min(2.0, domain.capacity \+ 0.02)  
    elif role \== "resource":  
        domain.capacity \= min(2.0, domain.capacity \+ 0.03)  
    elif role \== "boost":  
        domain.capacity \= min(2.5, domain.capacity \+ 0.05)  
        domain.load \= min(1.0, domain.load \+ 0.03)  
    elif role \== "shield":  
        domain.load \= max(0.0, domain.load \- 0.03)  
    elif role \== "absorb":  
        domain.load \= max(0.0, domain.load \- 0.05)  
    elif role \== "impact":  
        domain.load \= min(1.0, domain.load \+ 0.04)  
        if all\_domains:  
            for d in all\_domains:  
                if d is not domain:  
                    d.load \= max(0.0, d.load \- 0.02)  
    elif role \== "meta":  
        domain.capacity \= min(3.0, domain.capacity \+ 0.03)  
        domain.trust \= min(1.0, domain.trust \+ 0.02)  
    elif role \== "unify":  
        if all\_domains:  
            for d in all\_domains:  
                d.trust \= min(1.0, d.trust \+ 0.02)  
    elif role \== "connect":  
        domain.trust \= min(1.0, domain.trust \+ 0.04)  
    elif role \== "relate":  
        domain.trust \= min(1.0, domain.trust \+ 0.03)  
    elif role \== "broadcast":  
        if all\_domains:  
            for d in all\_domains:  
                d.trust \= min(1.0, d.trust \+ 0.01)  
    elif role \== "message":  
        domain.trust \= min(1.0, domain.trust \+ 0.02)  
    elif role \== "infra":  
        domain.health \= min(1.0, domain.health \+ 0.02)  
        domain.capacity \= min(2.0, domain.capacity \+ 0.02)  
    elif role \== "environment":  
        domain.health \= min(1.0, domain.health \+ 0.02)  
    elif role \== "creative":  
        domain.trust \= min(1.0, domain.trust \+ 0.02)  
    elif role \== "time":  
        \# tiny stabilizer  
        domain.load \= (domain.load \* 0.98) \+ 0.01  
    elif role \== "routing":  
        \# routing reduces effective load a bit  
        domain.load \= max(0.0, domain.load \- 0.02)  
    elif role \== "rescue":  
        domain.health \= min(1.0, domain.health \+ 0.04)  
    elif role \== "spawn":  
        domain.capacity \= min(2.5, domain.capacity \+ 0.04)  
    \# neutral: do nothing

def apply\_chain(domain, chain, all\_domains):  
    for spell in chain:  
        role \= SPELL\_ROLES.get(spell, DEFAULT\_ROLE)  
        apply\_spell(domain, role, all\_domains=all\_domains)

\# ENGINE  
\# \============================================================

def run\_hive\_sim(domains, bridges, hours=24):  
    print("=== INITIAL STATE \===")  
    for d in domains:  
        print(d)  
    print()

    for hour in range(hours):  
        print(f"\\n=== HOUR {hour} \===")

        \# Crisis/random progression  
        for d in domains:  
            d.load \= min(1.0, max(0.0, d.load \+ random.uniform(-0.02, 0.05)))  
            d.health \= max(0.0, d.health \- random.uniform(0.0, 0.01))

        \# Apply domain-local chains  
        for d in domains:  
            for chain in d.chains:  
                apply\_chain(d, chain, domains)

        \# Apply bridges (cross-domain effects)  
        for br in bridges:  
            a \= next(d for d in domains if d.name \== br.domain\_a)  
            b \= next(d for d in domains if d.name \== br.domain\_b)  
            for spell in br.chain:  
                role \= SPELL\_ROLES.get(spell, DEFAULT\_ROLE)  
                apply\_spell(a, role, all\_domains=domains)  
                apply\_spell(b, role, all\_domains=domains)

        \# Global Koantra/Equilibria-style balancer  
        max\_load \= max(d.load for d in domains)  
        min\_load \= min(d.load for d in domains)  
        if max\_load \- min\_load \> 0.4:  
            avg \= sum(d.load for d in domains) / len(domains)  
            for d in domains:  
                d.load \= (d.load \+ avg) / 2

        for d in domains:  
            print(d)

    print("\\n=== FINAL STATE \===")  
    for d in domains:  
        print(d)
